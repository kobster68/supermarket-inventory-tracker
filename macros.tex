% ---- qascenario: a six-part quality-attribute scenario as a boxed table ----------
% Usage:
%   \qascenario{AV-1}{source}{stimulus}{artifact}{environment}{response}{measure}
% The scenario identifier (first argument) is shown as the caption and links to the
% frozen label scn:<ID> declared in scenarios.tex.
\newcommand{\qascenario}[7]{%
  \par\medskip\noindent
  \begin{tabularx}{\linewidth}{|>{\bfseries}l|X|}
    \hline
    \multicolumn{2}{|l|}{\textbf{Quality-attribute scenario~\scnref{#1}}}\\
    \hline
    Source            & #2\\\hline
    Stimulus          & #3\\\hline
    Artifact          & #4\\\hline
    Environment       & #5\\\hline
    Response          & #6\\\hline
    Response measure  & #7\\\hline
  \end{tabularx}\par\medskip
}

% ---- Tactics table: header environment + one-row macro ---------------------------
% Columns: Scenario | Tactic (4th-edition name, page) | Realised in | Trade-off.
% Usage:
%   \begin{tacticstable}
%     \tacticrow{\scnref{AV-1}}{tactic}{realised in}{trade-off}
%   \end{tacticstable}
\newenvironment{tacticstable}{%
  \par\medskip\noindent
  \begin{tabularx}{\linewidth}{|l|X|X|X|}
  \hline
  \textbf{Scenario} & \textbf{Tactic (4e name, page)} & \textbf{Realised in} & \textbf{Trade-off}\\
  \hline
}{%
  \end{tabularx}\par\medskip
}
\newcommand{\tacticrow}[4]{#1 & #2 & #3 & #4\\\hline}

% ---- notapplicable: an unused SEI section shown as a deliberate decision ----------
\newenvironment{notapplicable}{%
  \par\medskip\noindent\textbf{Not applicable.}\ \ignorespaces
}{%
  \par\medskip
}

% ---- \todo: a visible marker; scripts/lint.sh --main rejects it on main -----------
\newcommand{\todo}[1]{%
  \par\medskip\noindent
  \fbox{\parbox{0.96\linewidth}{\textbf{TODO:}\ \itshape #1}}%
  \par\medskip
}
